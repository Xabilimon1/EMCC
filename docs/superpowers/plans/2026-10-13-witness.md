# witness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship witness 0.1.0, a live pane of Claude's work with proof marks, on top of emcc-core 0.2.0 (flags and `kindsOf`) and goalpost-guard 0.2.0 (reports and flags instead of asking when emcc-core is installed).

**Architecture:** emcc-core keeps the session's evidence log and gains a flag channel. goalpost-guard posts a flag for every change that lowers the bar. witness observes tool calls, prompts and turn ends, keeps goal, steps, edits, runs and ribbon segments in `$.state`, and computes each step's mark while drawing from that state plus emcc-core's checks and flags. Pure logic (goal parsing, phases, marks) lives in its own files with no `$`.

**Tech Stack:** Claude Code 2.1.287 mods (function hooks, TSX against the global `h`), `claude-code/testing`, `claude plugin validate --strict`, `tsc`, VHS for the recording.

**Spec:** `docs/superpowers/specs/2026-10-13-witness-design.md`. Deviations decided while planning, already reflected below:
- The `TaskCreate` id comes from the structured result (`result.task.id`), with the text `Task #N created` as fallback.
- Steps scroll with the pane's own scrolling (the engine scrolls panes); witness adds no ↑/↓ keys.
- A proven step shows `npm test passed 3m ago` (elapsed, from the session clock) instead of a wall-clock time, which would depend on the time zone.
- The line above the prompt reads `witness 2/4 proven` with a **Live** button; the EMCC badge reads `witness 2/4 proven, /live`. No ` · ` separators.
- `plugin.json` `dependencies` is an array of plugin names (`["emcc-core"]`); an object is rejected by the validator (checked).

## Conventions for every task

- Work in `~/Developer/EMCC` on `main` (the user merges without waiting; commit and push after each task).
- Commit messages: conventional (`feat(witness): …`), body says why, last line `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- A test only counts once it was seen failing. For new code the TDD order gives that. After a task's tests pass, break the line the test guards (one mutation per assertion family), run the test, see it fail, restore.
- Run from the repo root:
  - one mod's tests: `claude plugin test mods/<mod>`
  - validation: `claude plugin validate --strict .` and `claude plugin validate --strict mods/<mod>`
  - types: `npx -y -p typescript tsc -p .` from the repo root (the root tsconfig includes every mod and its contract; a per-mod tsc cannot see `$.emcc`)
- Never install the mods into the user's account. Real sessions use `--plugin-dir`.
- Colours are Anthropic's: sky `#6a9bcc`, olive `#788c5d`, clay `#d97757`, light `#e8e6dc`, stone `#b0aea5`.

`mods/witness/tests/fixtures/emcc.ts`:

```ts
import type { Plugin } from 'claude-code/testing'

/**
 * A stand-in for emcc-core 0.2.0. A test can't import the real module from
 * outside this plugin's folder, and a fixture runs on its own, so the test
 * drives it with commands: `emcc-check pass|fail <command>` records a test
 * run now, `emcc-flag <text>` raises a flag now, `emcc-badges` answers the
 * badges as JSON. `kindsOf` calls `npm test` and `tsc` checks.
 */
export const emcc: Plugin = {
  name: 'emcc-core',
  tier: 'user',
  register(on) {
    const evidence: { kind: string; at: number; command?: string; exitCode?: number; isOk?: boolean; file?: string }[] = []
    const flags: { source: string; rule: string; text: string; file?: string; at: number }[] = []
    const badges: unknown[] = []

    on('engine.create', async (_$, e, next) => {
      const beneath = await next(e)
      const added = {
        emcc: {
          profile: async () => 'standard' as const,
          isHalted: async () => false,
          badge: async (input: unknown) => {
            badges.push(input)
          },
          evidenceLatest: async () => null,
          evidenceSince: async ({ at }: { at: number }) => evidence.filter(item => item.at >= at),
          flag: async () => {},
          flagsSince: async ({ at }: { at: number }) => flags.filter(item => item.at >= at),
          kindsOf: async ({ command }: { command: string }) => (/\bnpm test\b|\btsc\b/.test(command) ? ['test'] : []),
        },
      }

      return { ...added, ...beneath }
    })
    on('command.run', { command: 'emcc-check' }, async ($, e) => {
      const [outcome = 'pass', ...words] = e.args.split(' ')
      const isOk = outcome === 'pass'
      evidence.push({ kind: 'test', at: await $.clock.now(), command: words.join(' '), exitCode: isOk ? 0 : 1, isOk })

      return { text: 'recorded' }
    })
    on('command.run', { command: 'emcc-flag' }, async ($, e) => {
      flags.push({ source: 'goalpost-guard', rule: 'tests.expected', text: e.args, at: await $.clock.now() })

      return { text: 'flagged' }
    })
    on('command.run', { command: 'emcc-badges' }, async () => ({ text: JSON.stringify(badges) }))
  },
}
```

import type { On } from 'claude-code'
import type { Engine, MockClock } from 'claude-code/testing'

export type Session = {
  clock: MockClock
  opened: { id: string; title?: string; focus?: boolean; closeOnEscape?: boolean }[]
  isWide: boolean
  failing: Set<string>
  isCwdBroken: boolean
  seenNow: unknown[]
}

/**
 * The engine beneath witness: a repo at /repo, a clock that moves only when
 * the test moves it, a pane that is placed while `isWide`, task ids counted
 * from 1, and commands or file paths that fail while listed in `failing`. A tool
 * called `Peek` records what witness holds as `now` while it runs.
 */
export function session(on: On): Session {
  const it: Session = {
    clock: mock.clock(on, { now: 0 }),
    opened: [],
    isWide: true,
    failing: new Set(),
    isCwdBroken: false,
    seenNow: [],
  }
  let nextId = 1
  on('session.cwd', () => {
    if (it.isCwdBroken) {
      throw new Error('cwd is gone')
    }

    return { value: '/repo' }
  })
  on('ui.open', (_$, e) => {
    it.opened.push(e as unknown as Session['opened'][number])

    return { value: it.isWide ? { isPlaced: true } : { isPlaced: false, reason: 'too narrow' } } as never
  })
  on('tool.call', async ($, e) => {
    const call = e as unknown as { tool: string; command?: string; subject?: string; taskId?: string }
    if (call.tool === 'TaskCreate') {
      const id = String(nextId)
      nextId += 1

      return { result: { task: { id, subject: call.subject ?? '' } }, text: `Task #${id} created successfully: ${call.subject ?? ''}` } as never
    }
    if (call.tool === 'TaskUpdate') {
      return { result: { success: true, taskId: call.taskId ?? '', updatedFields: ['status'] }, text: `Updated task #${call.taskId ?? ''}` } as never
    }
    if (call.tool === 'Peek') {
      it.seenNow.push((await $.state.get({ plugin: 'witness', key: 'now' })).value)
    }
    if (it.failing.has(call.command ?? (call as { file_path?: string }).file_path ?? '')) {
      return { isError: true, result: 'Exit code 1', text: 'Exit code 1\n1 failing' } as never
    }

    return { result: { stdout: '', stderr: '', interrupted: false }, text: 'ok' } as never
  })

  return it
}

/** Moves the clock `ms` on, then makes the call: the call's segment lasts `ms`. */
export async function call($: Engine, it: Session, input: Record<string, unknown>, ms = 1000) {
  await it.clock.advance(ms)

  return $.tool.call(input as never)
}

}

import type { WitnessCriterion } from '../types'

const DONE_WHEN = /\b(?:done when|hecho cuando)\b/i
const SENTENCE_END = /[.!?](?=\s|$)|\n/
const LONGEST = 100

export type ParsedGoal = { goal: string; criteria: WitnessCriterion[] }

export function collapsed(command: string): string {
  return command.trim().replace(/\s+/g, ' ')
}

export function sameCommand(a: string, b: string): boolean {
  return collapsed(a) === collapsed(b)
}

/** The first sentence of a prompt, at most 100 characters. */
export function goalFromPrompt(text: string): string {
  const trimmed = text.trim()
  const end = SENTENCE_END.exec(trimmed)
  const first = end === null ? trimmed : trimmed.slice(0, end[0] === '\n' ? end.index : end.index + 1).trim()

  return first.length > LONGEST ? `${first.slice(0, LONGEST - 1)}…` : first
}

/** `/goal` text: the goal, then criteria after "done when", split by `;`. */
export function parseGoal(text: string): ParsedGoal {
  const trimmed = text.trim()
  const match = DONE_WHEN.exec(trimmed)
  if (match === null) {
    return { goal: trimmed, criteria: [] }
  }
  const goal = trimmed.slice(0, match.index).replace(/[\s;,]+$/, '')
  const criteria = trimmed
    .slice(match.index + match[0].length)
    .split(';')
    .map(part => part.trim())
    .filter(part => part !== '')
    .map(part => {
      const inTicks = /`([^`]+)`/.exec(part)?.[1]

      return { text: part, command: inTicks === undefined ? null : collapsed(inTicks), isTicked: false }
    })

  return { goal, criteria }
}

import type { WitnessPhase, WitnessSegment } from '../types'

export const KEEP_SEGMENTS = 200

const EXPLORE_TOOLS = new Set(['Read', 'Grep', 'Glob', 'LS', 'NotebookRead', 'WebFetch', 'WebSearch'])
const EDIT_TOOLS = new Set(['Edit', 'Write', 'NotebookEdit'])
const READ_ONLY = /^\s*(?:rg|grep|ls|cat|head|tail|find|fd|tree|wc|git\s+(?:log|diff|show|status|blame))\b/

/** What witness knows of a finished tool call. `isCheck` comes from emcc-core's `kindsOf`. */
export type ToolFacts = { tool: string; command?: string; isCheck: boolean; isError: boolean }

export type PhaseStep = { phase: WitnessPhase | null; isDebugging: boolean }

/**
 * The phase a finished tool call belongs to, and whether Claude is debugging
 * after it: a failed check starts debugging, a passing one ends it. `null`
 * counts toward the phase in progress.
 */
export function phaseOf(facts: ToolFacts, isDebugging: boolean): PhaseStep {
  if (EDIT_TOOLS.has(facts.tool)) {
    return { phase: isDebugging ? 'debug' : 'implement', isDebugging }
  }
  if (EXPLORE_TOOLS.has(facts.tool)) {
    return { phase: 'explore', isDebugging }
  }
  if (facts.tool === 'Bash') {
    if (facts.isCheck) {
      return { phase: 'verify', isDebugging: facts.isError }
    }
    if (READ_ONLY.test(facts.command ?? '')) {
      return { phase: 'explore', isDebugging }
    }
  }

  return { phase: null, isDebugging }
}

/** The segments with this stretch added: the same phase extends the last one; past the cap the oldest two merge. */
export function appended(segments: readonly WitnessSegment[], phase: WitnessPhase, start: number, end: number): WitnessSegment[] {
  const last = segments.at(-1)
  let next =
    last !== undefined && last.phase === phase
      ? [...segments.slice(0, -1), { ...last, end: Math.max(last.end, end) }]
      : [...segments, { phase, start, end }]
  while (next.length > KEEP_SEGMENTS) {
    const [first, second, ...rest] = next
    if (first === undefined || second === undefined) {
      break
    }
    next = [{ phase: first.phase, start: first.start, end: second.end }, ...rest]
  }

  return next
}

export type Cell = { phase: WitnessPhase; chars: number }

/** The ribbon `width` characters wide, each phase's share by time. */
export function ribbonOf(segments: readonly WitnessSegment[], width: number): Cell[] {
  const total = segments.reduce((sum, segment) => sum + (segment.end - segment.start), 0)
  if (total <= 0 || width <= 0) {
    return []
  }
  const cells: Cell[] = []
  let elapsed = 0
  let used = 0
  for (const segment of segments) {
    elapsed += segment.end - segment.start
    const boundary = Math.round((elapsed / total) * width)
    const chars = boundary - used
    used = boundary
    if (chars <= 0) {
      continue
    }
    const last = cells.at(-1)
    if (last !== undefined && last.phase === segment.phase) {
      last.chars += chars
    } else {
      cells.push({ phase: segment.phase, chars })
    }
  }

  return cells
}

export function totalsOf(segments: readonly WitnessSegment[]): Partial<Record<WitnessPhase, number>> {
  const totals: Partial<Record<WitnessPhase, number>> = {}
  for (const segment of segments) {
    totals[segment.phase] = (totals[segment.phase] ?? 0) + (segment.end - segment.start)
  }

  return totals
}

export function durationText(ms: number): string {
  const seconds = Math.floor(ms / 1000)

  return seconds < 60 ? `${seconds}s` : `${Math.floor(seconds / 60)}m`
}

import type { WitnessCriterion, WitnessEdit, WitnessRun, WitnessStep } from '../types'
import { sameCommand } from './goal'

/** A check from emcc-core's evidence log (an `edit` entry is not one). */
export type Check = { kind: string; at: number; command?: string; isOk?: boolean }

/** A flag from emcc-core. */
export type Flag = { source: string; rule: string; text: string; file?: string; at: number }

export type Mark =
  | { kind: 'proven'; by: string; at: number }
  | { kind: 'unproven' }
  | { kind: 'stale'; editsAfter: number }
  | { kind: 'flagged'; text: string }
  | { kind: 'pending'; isActive: boolean }
  | { kind: 'unknown' }

export type CriterionMark =
  | { kind: 'proven'; at: number }
  | { kind: 'failing'; at: number }
  | { kind: 'stale'; editsAfter: number }
  | { kind: 'open' }
  | { kind: 'ticked' }

const PROVING = new Set(['test', 'build', 'typecheck'])

/**
 * A step's mark, computed from what happened, never stored. `checks` and
 * `flags` are null when emcc-core could not be asked.
 */
export function markOf(
  step: WitnessStep,
  edits: readonly WitnessEdit[],
  checks: readonly Check[] | null,
  flags: readonly Flag[] | null,
): Mark {
  const begin = step.startedAt ?? step.createdAt
  const end = step.completedAt ?? Number.POSITIVE_INFINITY
  const flag = (flags ?? []).filter(one => one.at >= begin && one.at <= end).at(-1)
  if (flag !== undefined) {
    return { kind: 'flagged', text: flag.text }
  }
  if (step.status !== 'completed') {
    return { kind: 'pending', isActive: step.status === 'in_progress' }
  }
  if (checks === null) {
    return { kind: 'unknown' }
  }
  const proof = checks.filter(check => PROVING.has(check.kind) && check.isOk === true && check.at >= begin).at(-1)
  if (proof === undefined) {
    return { kind: 'unproven' }
  }
  const files = new Set(edits.filter(one => step.id !== null && one.stepId === step.id).map(one => one.file))
  const editsAfter = edits.filter(one => one.at > proof.at && (files.size === 0 || files.has(one.file))).length

  return editsAfter === 0 ? { kind: 'proven', by: proof.command ?? proof.kind, at: proof.at } : { kind: 'stale', editsAfter }
}

/** A `/goal` criterion's mark: its command's newest run, against every edit after it. */
export function criterionMarkOf(
  criterion: WitnessCriterion,
  runs: readonly WitnessRun[],
  edits: readonly WitnessEdit[],
): CriterionMark {
  if (criterion.command === null) {
    return criterion.isTicked ? { kind: 'ticked' } : { kind: 'open' }
  }
  const command = criterion.command
  const run = runs.filter(one => sameCommand(one.command, command)).at(-1)
  if (run === undefined) {
    return { kind: 'open' }
  }
  if (!run.isOk) {
    return { kind: 'failing', at: run.at }
  }
  const editsAfter = edits.filter(one => one.at > run.at).length

  return editsAfter === 0 ? { kind: 'proven', at: run.at } : { kind: 'stale', editsAfter }
}

async function before($: EngineInterface, call: ToolCall): Promise<number> {
  const at = await $.clock.now()
  const cwd = await $.session.cwd()
  await closeWaiting($, at)
  if ((await read($, cursor)) === null) {
    await update($, cursor, () => at)

  await update($, now, () => ({ label: labelOf(call, cwd), since: at }))
  startBeat($)
  if (!(await read($, hasOpened))) {
    await update($, hasOpened, () => true)
    await openPane($, false)
  }

  return at
}

async function after($: EngineInterface, call: ToolCall, ran: Ran): Promise<void> {
  const at = await $.clock.now()
  endBeat()
  await update($, now, () => null)
  if (ran.deny !== undefined) {
    return
  }
  const isFailed = ran.isError === true
  const checked = call.tool === 'Bash' && call.command !== undefined && (await isCheck($, call.command))
  const step = phaseOf({ tool: call.tool, command: call.command, isCheck: checked, isError: isFailed }, await read($, isDebugging))
  await update($, isDebugging, () => step.isDebugging)
  const phase = step.phase ?? (await read($, segments)).at(-1)?.phase ?? 'explore'
  const from = (await read($, cursor)) ?? at
  await update($, segments, list => appended(list, phase, from, at))
  await update($, cursor, () => at)

  const file = call.file_path ?? call.notebook_path
  if (EDIT_TOOLS.has(call.tool) && file !== undefined && !isFailed) {
    const active = (await read($, steps)).filter(one => one.status === 'in_progress').at(-1)
    await update($, edits, list => [...list, { file, at, stepId: active?.id ?? null }].slice(-KEEP_EDITS))
  }
  if (call.tool === 'Bash' && call.command !== undefined) {
    const command = collapsed(call.command)
    await update($, runs, list => [...list, { command, at, isOk: !isFailed }].slice(-KEEP_RUNS))
  }
  if (call.tool === 'TaskCreate' && !isFailed) {
    const structured = (ran.result as { task?: { id?: unknown } } | undefined)?.task?.id
    const id = typeof structured === 'string' ? structured : (/Task #(\S+) created/.exec(ran.text ?? '')?.[1] ?? null)

    await update($, steps, list => [...list, added])
  }
  if (call.tool === 'TaskUpdate' && !isFailed && call.taskId !== undefined) {
    const id = call.taskId
    await update($, steps, list =>
      list.flatMap(one => {
        if (one.id !== id) {
          return [one]
        }
        if (call.status === 'deleted') {
          return []
        }
        const status = call.status === 'pending' || call.status === 'in_progress' || call.status === 'completed' ? call.status : one.status

            subject: call.subject ?? one.subject,
            status,
            startedAt: status === 'in_progress' && one.startedAt === null ? at : one.startedAt,
            completedAt: status === 'completed' ? (one.completedAt ?? at) : null,
          },
        ]
      }),
    )
  }
  await refreshBadge($)
}

export const register: Register = on => {
  on('session.start', async ($, e, next) => {
    await $.command.register({ name: 'live', description: 'witness: open the live view, or "resume" after pausing it' })
    await $.command.register({ name: 'goal', description: 'witness: set the goal, with criteria after "done when"; empty clears it' })

    return next(e)
  })

  on('prompt.submit', async ($, e, next) => {
    if (await read($, isPaused)) {
      return next(e)
    }
    try {
      const at = await $.clock.now()
      await closeWaiting($, at)
      await update($, cursor, () => at)
      if (!e.text.trimStart().startsWith('/')) {
        const fromPrompt = goalFromPrompt(e.text)
        await update($, promptGoal, () => fromPrompt)
        if (!(await read($, isGoalSet))) {
          await update($, goal, () => fromPrompt)
        }
      }
    } catch {
      await lose($, 'prompt.submit')
    }

    return next(e)
  })

  on('tool.call', async ($, e, next) => {
    if (await read($, isPaused)) {
      return next(e)
    }
    const call = e as unknown as ToolCall
    try {
      await before($, call)
    } catch {
      await lose($, 'tool.call')
    }
    const ran = await next(e)
    try {
      await after($, call, ran as unknown as Ran)
    } catch {
      await lose($, 'tool.call')
    }

    return ran
  })

  on('turn.complete', async ($, e, next) => {
    try {
      const at = await $.clock.now()
      endBeat()
      await update($, now, () => null)
      await update($, waitingSince, () => at)
    } catch {
      await lose($, 'turn.complete')
    }

    return next(e)
  })

  on('command.run', { command: 'live' }, async ($, e) => {
    if (e.args.trim() === 'resume') {
      await update($, isPaused, () => false)

      return { text: 'witness is watching again.' }
    }
    const placed = await openPane($, true)
    await refreshBadge($)

    return { text: placed ? 'witness opened.' : await summaryText($) }
  })

  on('command.run', { command: 'goal' }, async ($, e) => {
    const text = e.args.trim()
    if (text === '') {
      const fallback = await read($, promptGoal)
      await update($, goal, () => fallback)
      await update($, isGoalSet, () => false)
      await update($, criteria, () => [])

      return { text: 'Goal cleared; it follows your prompt again.' }
    }
    const parsed = parseGoal(text)
    await update($, goal, () => parsed.goal)
    await update($, isGoalSet, () => true)
    await update($, criteria, () => parsed.criteria)
    const count = parsed.criteria.length

    return { text: count === 0 ? 'Goal set.' : `Goal set with ${count} ${count === 1 ? 'criterion' : 'criteria'}.` }
  })
}

type Look = { glyph: string; color?: string; note: string }

function stepLook(mark: Mark, nowAt: number): Look {
  switch (mark.kind) {
    case 'proven':
      return { glyph: '✓', color: OLIVE, note: `${mark.by} passed ${durationText(nowAt - mark.at)} ago` }
    case 'unproven':
      return { glyph: '○', color: STONE, note: 'no proof' }
    case 'stale':
      return { glyph: '◌', color: STONE, note: `proven before ${mark.editsAfter} ${mark.editsAfter === 1 ? 'edit' : 'edits'}` }
    case 'flagged':
      return { glyph: '▲', color: CLAY, note: mark.text }
    case 'pending':
      return { glyph: '·', color: mark.isActive ? undefined : STONE, note: mark.isActive ? 'in progress' : '' }
    case 'unknown':
      return { glyph: '○', color: STONE, note: '' }
  }
}

function criterionLook(mark: CriterionMark, nowAt: number): Look {
  switch (mark.kind) {
    case 'proven':
      return { glyph: '✓', color: OLIVE, note: `passed ${durationText(nowAt - mark.at)} ago` }
    case 'failing':
      return { glyph: '✗', color: CLAY, note: `failed ${durationText(nowAt - mark.at)} ago` }
    case 'stale':
      return { glyph: '◌', color: STONE, note: `proven before ${mark.editsAfter} ${mark.editsAfter === 1 ? 'edit' : 'edits'}` }
    case 'ticked':
      return { glyph: '✓', color: OLIVE, note: 'ticked by you' }
    case 'open':
      return { glyph: '·', color: STONE, note: '' }
  }
}
